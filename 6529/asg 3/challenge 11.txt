#include <stdio.h>

int main() {
    FILE *fp;

    // Write to file
    fp = fopen("data.txt", "w");
    if (fp == NULL) {
        printf("Error opening file!\n");
        return 1;
    }
    fprintf(fp, "Hello, this is a test.\n");
    fclose(fp);

    // Append to file
    fp = fopen("data.txt", "a");
    if (fp == NULL) {
        printf("Error opening file!\n");
        return 1;
    }
    fprintf(fp, "Appending new line.\n");
    fclose(fp);

    // Read from file
    fp = fopen("data.txt", "r");
    if (fp == NULL) {
        printf("Error opening file!\n");
        return 1;
    }
    char buffer[100];
    while (fscanf(fp, "%99[^\n]\n", buffer) == 1) {
        printf("%s\n", buffer);
    }
    fclose(fp);

    return 0;
}
