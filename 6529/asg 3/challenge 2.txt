// Include the standard input-output header file
#include <stdio.h>

// The main function: execution starts here
int main() {
    // Declare variables to store user input
    char name[50];   // String for name
    int age;         // Integer for age
    float salary;    // Floating-point number for salary

    // Ask the user for their name
    printf("Enter your name: ");
    scanf("%s", name);  // %s is used for strings (no spaces)

    // Ask the user for their age
    printf("Enter your age: ");
    scanf("%d", &age);  // %d is used for integers

    // Ask the user for their salary
    printf("Enter your salary: ");
    scanf("%f", &salary);  // %f is used for floats

    // Print the collected information neatly
    printf("\n--- User Information ---\n");
    printf("Name: %s\n", name);
    printf("Age: %d\n", age);
    printf("Salary: %.2f\n", salary);  // %.2f prints salary with 2 decimal places

    // Return 0 to indicate successful program termination
    return 0;
}
