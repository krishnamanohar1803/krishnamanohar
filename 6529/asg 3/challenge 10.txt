#include <stdio.h>

// Function to swap two numbers using pointers
void swap(int *a, int *b) {
    int temp = *a;
    *a = *b;
    *b = temp;
}

int main() {
    int x = 10, y = 20;
    int arr[5] = {1, 2, 3, 4, 5};
    int *ptr;

    // Print addresses and values
    printf("Address of x: %p, Value: %d\n", &x, x);
    printf("Address of y: %p, Value: %d\n", &y, y);

    // Swap using pointers
    swap(&x, &y);
    printf("After swap: x = %d, y = %d\n", x, y);

    // Traverse array using pointers (no indexing)
    ptr = arr;
    printf("Array elements: ");
    for (int i = 0; i < 5; i++) {
        printf("%d ", *(ptr + i));
    }
    printf("\n");

    return 0;
}
